function updateMenstrualCycle() { const START_WORD = "生理開始"; const END_WORD = "生理終了"; const COLOR_PERIOD = "1"; const COLOR_OVULATION = "6"; const COLOR_FERTILITY = "4"; const calendar = CalendarApp.getDefaultCalendar(); const now = new Date(); now.setHours(0,0,0,0); const userEmail = Session.getActiveUser().getEmail(); if (!userEmail) return; const historyStart = new Date(now.getTime() - (180*24*60*60*1000)); let startEvents = calendar.getEvents(historyStart, now, { search: START_WORD }); let endEvents = calendar.getEvents(historyStart, now, { search: END_WORD }); if (startEvents.length < 2) { sendMailOnce("DATA_ERROR","【通知】データ不足", "生理開始が2件以上必要です。"); return; } startEvents.sort((a,b)=>a.getStartTime()-b.getStartTime()); endEvents.sort((a,b)=>a.getStartTime()-b.getStartTime()); // ===== 周期計算（直近3回） ===== let cycles=[]; for(let i=startEvents.length-3;i=0){ let diff=startEvents[i+1].getStartTime()-startEvents[i].getStartTime(); let days=diff/(1000*60*60*24); if(days>15 && days<50) cycles.push(days); } } if(cycles.length===0) return; const avgCycle=Math.round(cycles.reduce((a,b)=>a+b)/cycles.length); // ===== 異常周期アラート ===== if (avgCycle < 21 || avgCycle > 39) { sendMailOnce("CYCLE_ALERT", "【注意】周期が不安定の可能性", "現在の平均周期は " + avgCycle + " 日です。\n周期が通常範囲外の可能性があります。体調に注意してください。"); } // ===== 生理日数計算 ===== let periodLengths=[]; startEvents.forEach(start=>{ let startDate=start.getStartTime(); let endMatch=endEvents.find(e=> Math.abs(e.getStartTime()-startDate)<10*24*60*60*1000 ); if(endMatch){ let diff=endMatch.getStartTime()-startDate; let days=diff/(1000*60*60*24); if(days>=2 && days<=10) periodLengths.push(days); } }); const avgPeriodLength = periodLengths.length>0 ? Math.round(periodLengths.reduce((a,b)=>a+b)/periodLengths.length) : 5; const latestStart=new Date(startEvents[startEvents.length-1].getStartTime()); const nextStart=new Date(latestStart); nextStart.setDate(latestStart.getDate()+avgCycle); const nextEnd=new Date(nextStart); nextEnd.setDate(nextStart.getDate()+avgPeriodLength); const ovulation=new Date(nextStart); ovulation.setDate(nextStart.getDate()-14); const fertilityStart=new Date(ovulation); fertilityStart.setDate(ovulation.getDate()-3); const fertilityEnd=new Date(ovulation); fertilityEnd.setDate(ovulation.getDate()+1); // ===== 未来予測削除 ===== const futureSearch=new Date(now.getTime()+60*24*60*60*1000); calendar.getEvents(now,futureSearch).forEach(e=>{ if(e.getTitle().includes("【予測】")) e.deleteEvent(); }); // ===== 予測作成 ===== calendar.createAllDayEvent("【予測】次回生理開始（"+avgCycle+"日周期）",nextStart).setColor(COLOR_PERIOD); calendar.createAllDayEvent("【予測】次回生理終了",nextStart,nextEnd).setColor(COLOR_PERIOD); calendar.createAllDayEvent("【予測】排卵推定日",ovulation).setColor(COLOR_OVULATION); calendar.createAllDayEvent("【予測】妊娠可能性が高い期間",fertilityStart,fertilityEnd).setColor(COLOR_FERTILITY); function isSameDay(d1,d2){ return d1.getFullYear()===d2.getFullYear() && d1.getMonth()===d2.getMonth() && d1.getDate()===d2.getDate(); } function sendMailOnce(key,subject,body){ const props=PropertiesService.getScriptProperties(); const last=props.getProperty(key); if(last!==now.toDateString()){ GmailApp.sendEmail(userEmail,subject,body); props.setProperty(key,now.toDateString()); } } let weekBefore=new Date(nextStart); weekBefore.setDate(nextStart.getDate()-7); if(isSameDay(now,weekBefore)){ sendMailOnce("PERIOD_WEEK", "【通知】生理予定日一週間前です", "生理予定日一週間前です。"); } let after7=new Date(nextStart); after7.setDate(nextStart.getDate()+7); if(now>after7){ sendMailOnce("PERIOD_MISS", "【確認】生理開始日が入力されていません", "生理開始日が入力されていません。"); } let ovuBefore=new Date(ovulation); ovuBefore.setDate(ovulation.getDate()-3); if(isSameDay(now,ovuBefore)){ sendMailOnce("OVU_NOTICE", "【通知】妊娠可能性が高い期間に入ります", "妊娠可能性が高い期間に入ります。"); } if(isSameDay(now,ovulation)){ sendMailOnce("OVU_DAY", "【通知】本日が排卵予測日です", "本日が排卵予測日です。"); } const props=PropertiesService.getScriptProperties(); let sheetId=props.getProperty("SHEET_ID"); if(!sheetId){ const ss=SpreadsheetApp.create("生理周期ログ"); sheetId=ss.getId(); props.setProperty("SHEET_ID",sheetId); } const sheet=SpreadsheetApp.openById(sheetId).getActiveSheet(); sheet.appendRow([now,avgCycle,avgPeriodLength,nextStart,nextEnd,ovulation]); console.log("異常周期アラート入り精密版 完了"); }
